def buscar(mapa, entrega):
    origem = entrega["origem"]
    destino = entrega["destino"]
    largura = len(mapa[0])
    fila = [origem]
    inicio = 0
    pais = {}
    mapa[origem[0]][origem[1]] = 0
    achou = False
    while inicio < len(fila) and not achou:
        atual = fila[inicio]
        inicio = inicio + 1
        if atual[0] == destino[0] and atual[1] == destino[1]:
            achou = True
        else:
            for passo in [[-1, 0], [1, 0], [0, -1], [0, 1]]:
                linha = atual[0] + passo[0]
                coluna = atual[1] + passo[1]
                if linha >= 0 and linha < len(mapa) and coluna >= 0 and coluna < largura:
                    if mapa[linha][coluna] == 1:
                        proxima = [linha, coluna]
                        pais[linha * largura + coluna] = atual
                        fila.append(proxima)
                        mapa[linha][coluna] = 0
    rota = []
    if achou:
        atual = destino
        while atual[0] != origem[0] or atual[1] != origem[1]:
            rota = [atual] + rota
            atual = pais[atual[0] * largura + atual[1]]
        rota = [origem] + rota
    return rota


produtos = tabela_produtos()
while True:
    if sensor("cliente_na_tela"):
        pedido = input()
        regras = promocoes()
        total = 0
        for codigo in pedido["carrinho"]:
            quantidade = pedido["carrinho"][codigo]
            preco = produtos[codigo]["preco"]
            subtotal = quantidade * preco
            for regra in regras:
                if regra["produto"] == codigo:
                    if regra["tipo"] == "leve_pague":
                        pagas = (quantidade // regra["leve"]) * regra["pague"] + quantidade % regra["leve"]
                        subtotal = pagas * preco
                    elif regra["tipo"] == "percentual":
                        subtotal = subtotal * (1 - regra["desconto"])
            total = total + subtotal
        total = ((total * 100 + 0.5) // 1) / 100
        if "pagamento" in pedido:
            troco = ((pedido["pagamento"] - total) * 100 + 0.5) // 1 / 100
            send(total, troco)
        else:
            send(total)
    estoque = get_stock()
    for codigo in produtos:
        produto = produtos[codigo]
        lote = produto["lote"]
        if estoque[codigo] < produto["minimo"]:
            if estoque[codigo] + lote <= produto["capacidade"]:
                if lote * produto["custo"] <= get_budget():
                    resultado = buy_stock({codigo: lote})
                    if not resultado["ok"]:
                        print(resultado["mensagem"])
    entrega = entrega_atual()
    if entrega:
        mapa = mapa_entrega()
        rota = buscar(mapa, entrega)
        if rota:
            enviar_rota(rota)
    wait(0.2)
